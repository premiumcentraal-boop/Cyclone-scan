cmake_minimum_required(VERSION 3.22)
project(dtnative C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(ANDROID)
    add_library(dtnative SHARED dt_checks.c dt_io.c dt_jni.c dt_sha256.c dt_elf.c)
    target_link_libraries(dtnative PRIVATE dl)
    target_compile_options(dtnative PRIVATE -Wall -Wextra -Werror -fvisibility=hidden -fstack-protector-strong)
    target_link_options(dtnative PRIVATE -Wl,-z,relro,-z,now,-z,max-page-size=16384)
else()
    enable_testing()
    add_library(dt_fixture SHARED tests/dt_fixture.c)
    add_executable(dt_elf_test dt_elf.c dt_sha256.c dt_io.c tests/dt_elf_test.c)
    target_include_directories(dt_elf_test PRIVATE .)
    target_compile_options(dt_elf_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(dt_elf_test PRIVATE dl dt_fixture)
    add_test(NAME owned_elf_integrity COMMAND dt_elf_test)
    add_executable(dt_checks_test dt_checks.c tests/dt_checks_test.c)
    target_include_directories(dt_checks_test PRIVATE .)
    target_compile_options(dt_checks_test PRIVATE -Wall -Wextra -Werror)
    if(DT_SANITIZE)
        target_compile_options(dt_checks_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(dt_checks_test PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME native_checks COMMAND dt_checks_test)
    add_executable(dt_io_test dt_io.c tests/dt_io_test.c)
    target_include_directories(dt_io_test PRIVATE .)
    target_compile_options(dt_io_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME raw_io COMMAND dt_io_test)
endif()
